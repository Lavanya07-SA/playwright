03	dblclick()      	double clicks the selected element

05	title()	                gets main tab title

06	url()    	        gets website link

07	goto()          	navigates to website

08	goBack()	        goes back from current page

09	goForward()     	goes forward from current page

24	press()         	hitting physical button like enter in keyboard

25	isdisabled()    	checks whether disabled

26	isenabled()	        checks whether enabled

27	ischecked()	        checks whether checkbox or radio button element is selected

30	screenshot()	        captures image of current page or specific element

31	hover()	                moves mouse pointer directly over element without clicking it
